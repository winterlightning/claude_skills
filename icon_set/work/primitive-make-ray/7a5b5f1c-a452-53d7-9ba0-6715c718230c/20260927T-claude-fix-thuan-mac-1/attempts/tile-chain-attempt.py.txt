"""Deep learning amis: a rounded app tile holding a vertical chain of three
network nodes joined by short links.

Revision (disapproved, reason not recorded): the rejected drawing squeezed the
three nodes into touching circles that fused into one blob, so the chain of
separate linked nodes in the original did not read. The nodes are now separate
rings with visible links between them.

Symbol plan: mirror axis x=24. Tile: rounded square (6,6)-(42,42), corner radius
4. Nodes: three radius-3 rings on x=24 at y=12, 24 and 36 (the small-circle
exception keeps their counters legal). Links: straight segments between the
rings, and the chain runs on to the tile's top and bottom edges so every node is
held by a link rather than floating closer than 8 to the tile.
Omissions: the gap between the chain ends and the tile in the reference (a
floating three-node chain needs 34 units of height; the tile gives 20).
Lucide construction: 'square' rounded tile; 'git-commit-vertical' ring-on-line.
Keyshape SQUARE: centerline (6,6)-(42,42).
"""
from icon_set.model.keyshapes import Keyshape
from icon_set.model.icons.solo._base import Solo48

SOURCE_ICON_ID = "7a5b5f1c-a452-53d7-9ba0-6715c718230c"
SOURCE_PATH = "icon_set/work/primitive-fix-thuan/solo__deep-learning-amis/20260926T182653Z-thuan-mac-1/reference/deep learning amis_7a5b5f1c-a452-53d7-9ba0-6715c718230c.svg"
AUTHOR = "claude-opus-5-5"

CX, NODE_R = 24, 3
NODES = (14, 24, 34)


class DeepLearningAmis(Solo48):
    icon_id = "deep-learning-amis"
    keyshape = Keyshape.SQUARE
    semantic_role = "MAIN"
    semantic_kind = "noun"
    category = "technology"
    aliases = ("neural-chain", "deep-learning")
    keywords = ("deep", "learning", "neural", "network", "nodes", "ai", "machine", "learning")

    def build(self) -> None:
        left, top, right, bottom, radius = 6, 6, 42, 42, 4
        self.add_line("outline-top-left", (left + radius, top), (CX, top))
        self.add_line("outline-top-right", (CX, top), (right - radius, top))
        self.add_arc("outline-tr", (right - radius, top), (right, top + radius), radius_x=radius)
        self.add_line("outline-right", (right, top + radius), (right, bottom - radius))
        self.add_arc("outline-br", (right, bottom - radius), (right - radius, bottom), radius_x=radius)
        self.add_line("outline-bottom-right", (right - radius, bottom), (CX, bottom))
        self.add_line("outline-bottom-left", (CX, bottom), (left + radius, bottom))
        self.add_arc("outline-bl", (left + radius, bottom), (left, bottom - radius), radius_x=radius)
        self.add_line("outline-left", (left, bottom - radius), (left, top + radius))
        self.add_arc("outline-tl", (left, top + radius), (left + radius, top), radius_x=radius)
        self.add_contour("outline", "outline-top-left", "outline-top-right", "outline-tr", "outline-right",
                         "outline-br", "outline-bottom-right", "outline-bottom-left", "outline-bl",
                         "outline-left", "outline-tl", closed=True)

        stops = [top]
        for i, y in enumerate(NODES):
            n = f"node-{i}"
            self.add_arc(f"{n}-a", (CX, y - NODE_R), (CX + NODE_R, y), radius_x=NODE_R)
            self.add_arc(f"{n}-b", (CX + NODE_R, y), (CX, y + NODE_R), radius_x=NODE_R)
            self.add_arc(f"{n}-c", (CX, y + NODE_R), (CX - NODE_R, y), radius_x=NODE_R)
            self.add_arc(f"{n}-d", (CX - NODE_R, y), (CX, y - NODE_R), radius_x=NODE_R)
            self.add_contour(n, f"{n}-a", f"{n}-b", f"{n}-c", f"{n}-d", closed=True)
            stops += [y - NODE_R, y + NODE_R]
        stops.append(bottom)
        owners = ["outline"] + [f"node-{i}" for i in range(3) for _ in (0, 1)] + ["outline"]
        for j in range(0, len(stops), 2):
            link = f"link-{j // 2}"
            self.add_line(link, (CX, stops[j]), (CX, stops[j + 1]))
            self.relate("connect", link, owners[j])
            self.relate("connect", link, owners[j + 1])
