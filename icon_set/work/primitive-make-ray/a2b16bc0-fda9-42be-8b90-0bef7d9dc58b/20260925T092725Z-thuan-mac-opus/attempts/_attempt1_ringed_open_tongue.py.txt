    semantic_kind = "noun"
    category = "emoji"

    def build(self) -> None:

        self.add_arc("face-2", (44, 24), (4, 24), radius_x=20)

        self.add_bezier("mouth-1", (16, 27), ((17, 29.5), (18.5, 31), left))

        self.add_bezier("mouth-3", right, ((29.5, 31), (31, 29.5), (32, 27)))
        self.add_contour("mouth", "mouth-1", "mouth-2", "mouth-3")
