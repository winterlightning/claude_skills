class DogPawPrintOvalToes(Solo48):
    icon_id = "dog-paw-print-oval-toes"
    keyshape = Keyshape.SQUARE
    semantic_role = "MAIN"
    semantic_kind = "noun"
    category = "animals/pet"
    aliases = ("dog paw", "paw print", "pawprint")
    keywords = ("paw", "print", "dog", "cat", "pet", "animal", "footprint", "track", "vet")

    def ring(self, name, cx, cy, rx, ry):
        pts = [(cx - rx, cy), (cx, cy - ry), (cx + rx, cy), (cx, cy + ry)]
        names = tuple(f"{name}-{q}" for q in ("nw", "ne", "se", "sw"))
        for i, n in enumerate(names):
            self.add_arc(n, pts[i], pts[(i + 1) % 4], radius_x=rx, radius_y=ry)
        self.add_contour(name, *names, closed=True)

    def build(self) -> None:
        for name, (x, y) in (("toe-inner-left", (16, 11)), ("toe-inner-right", (32, 11)),
                             ("toe-outer-left", (9, 24)), ("toe-outer-right", (39, 24))):
            self.ring(name, x, y, 3, 5)
        self.add_arc("pad-top", (20, 26), (28, 26), radius_x=5)
        self.add_line("pad-flank-right", (28, 26), (34, 34))
        self.add_arc("pad-corner-right", (34, 34), (30, 42), radius_x=5)
        self.add_line("pad-bottom", (30, 42), (18, 42))
        self.add_arc("pad-corner-left", (18, 42), (14, 34), radius_x=5)
        self.add_line("pad-flank-left", (14, 34), (20, 26))
        self.add_contour("pad", "pad-top", "pad-flank-right", "pad-corner-right", "pad-bottom",
                         "pad-corner-left", "pad-flank-left", closed=True)
