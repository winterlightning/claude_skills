Symbol plan: one closed silhouette. The handle is an axis-aligned rounded square

Lucide construction: 'hand saw' style - straight back, stepped teeth, closed grip; no exact

Keyshape SQUARE: centerline 6..42 (tip y 6, tooth x 6, handle right/bottom 42).
"""

from icon_set.model.icons.solo._base import Solo48

class DiagonalHandSaw(Solo48):
    icon_id = "diagonal-hand-saw"
    keyshape = Keyshape.SQUARE
    semantic_role = "MAIN"
    semantic_kind = "noun"
    category = "objects/tool"
    aliases = ("tools saw", "saw", "handsaw")
    keywords = ("saw", "tool", "carpentry", "woodwork", "cut", "diy", "construction", "hardware")

    def build(self) -> None:
        lo, hi, r = 20, 42, 4
        teeth = [(6, 14), (13, 14), (13, 21), (20, 21)]
        names = []

        def line(n, a, b):
            self.add_line(n, a, b)
            names.append(n)

        def arc(n, a, b):
            self.add_arc(n, a, b, radius_x=r)
            names.append(n)

        line("back", (17, 6), (31, lo))
        line("handle-top", (31, lo), (hi - r, lo))
        arc("handle-corner-ne", (hi - r, lo), (hi, lo + r))
        line("handle-right", (hi, lo + r), (hi, hi - r))
        arc("handle-corner-se", (hi, hi - r), (hi - r, hi))
        line("handle-bottom", (hi - r, hi), (lo + r, hi))
        arc("handle-corner-sw", (lo + r, hi), (lo, hi - r))
        line("handle-left", (lo, hi - r), teeth[-1])
        rev = list(reversed(teeth))
        for i in range(len(rev) - 1):
            line(f"teeth-{i + 1}", rev[i], rev[i + 1])
        line("tip", teeth[0], (17, 6))
        self.add_contour("saw", *names, closed=True)
        cx = cy = 31
        pts = [(cx - 2, cy), (cx, cy - 2), (cx + 2, cy), (cx, cy + 2)]
        hole = ("hole-nw", "hole-ne", "hole-se", "hole-sw")
        for i, n in enumerate(hole):
            self.add_arc(n, pts[i], pts[(i + 1) % 4], radius_x=2)
        self.add_contour("grip-hole", *hole, closed=True)
