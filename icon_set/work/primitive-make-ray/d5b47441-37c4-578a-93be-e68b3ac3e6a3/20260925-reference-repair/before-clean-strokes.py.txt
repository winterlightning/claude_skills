All defining parts retained. References: supplied wine barrel and Lucide barrel original/atomic geometry.'
from icon_set.model.keyshapes import Keyshape
from icon_set.model.icons.solo._base import Solo48
AUTHOR='gpt-6'
SOURCE_ICON_ID='d5b47441-37c4-578a-93be-e68b3ac3e6a3'
SOURCE_PATH='pictographic-primitives/drinks/wine barrel_d5b47441-37c4-578a-93be-e68b3ac3e6a3.svg'
class Drawing(Solo48):
    icon_id='wooden-barrel-with-two-hoops-and-curved-staves'
    keyshape=Keyshape.SQUARE
    category='objects/drink'
    semantic_role='MAIN'
    semantic_kind='noun'
    aliases=()
    keywords=()
    def build(self):
        # Two mirrored bowed staves follow the convex barrel walls.
        for side in ('left','right'):
            flip=lambda p: p if side=='left' else (48-p[0],p[1])
            for name,pts in [('wall',[(10,6),(7,16),(6,24),(7,32),(10,42)]),('stave',[(18,6),(16,16),(15,24),(16,32),(18,42)])]:
                controls=([(8,9),(7,12),(6,19),(6,21),(6,27),(6,29),(7,36),(8,39)] if name=='wall' else [(17,9),(16,12),(15,19),(15,21),(15,27),(15,29),(16,36),(17,39)])
                for i in range(4):
                    self.add_bezier(f'{name}-{side}-{i}',flip(pts[i]),(flip(controls[2*i]),flip(controls[2*i+1]),flip(pts[i+1])))
                self.add_contour(name+'-'+side,*(f'{name}-{side}-{i}' for i in range(4)))
        for name,y in [('top',6),('bottom',42)]:
            self.add_polyline(name,(10,y),(18,y),(30,y),(38,y))
            for part in ('wall-left','wall-right','stave-left','stave-right'):self.relate('connect',name,part)
        for y in (16,32):
            self.add_polyline('hoop-'+str(y),(7,y),(16,y),(32,y),(41,y))
            for part in ('wall-left','wall-right','stave-left','stave-right'):self.relate('connect','hoop-'+str(y),part)
