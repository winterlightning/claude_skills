"""Grinning face with heart eyes: a round face, two heart-shaped eyes and a
wide open D-shaped grin.

Symbol plan: face = radius-20 circle about (24,24). Both eyes are one heart
definition (bottom point, straight-ish sides, two lobes meeting in a cusp
at the dip) instanced on the vertical axis x = 24 at x = 17 and 31, 8 apart.
The grin is a closed D (straight top, one cubic bowl). Every feature stays
within radius 12 of the centre so it clears the rim by 8.
Mirror-symmetric about x = 24.
Keyshape CIRCLE, radius 20 about (24,24).
Lucide construction: smile / laugh (circle face, features inside, D grin)
and heart (two lobes into a point).
"""
from icon_set.model.keyshapes import Keyshape
from icon_set.model.icons.solo._base import Solo48

SOURCE_ICON_ID = "bc11fb2a-33c3-4ee2-90ab-066b5771ea82"
SOURCE_PATH = "icon_set/work/primitive-fix-thuan/solo__grinning-face-with-heart-eyes/20260925T092530Z-thuan-mac/reference/face grin hearts_bc11fb2a-33c3-4ee2-90ab-066b5771ea82.svg"
AUTHOR = "claude-opus-5-5"


class GrinningFaceWithHeartEyes(Solo48):
    icon_id = "grinning-face-with-heart-eyes"
    keyshape = Keyshape.CIRCLE
    semantic_role = "MAIN"
    semantic_kind = "noun"
    category = "emoji"
    aliases = ("heart eyes", "in love", "smiling face with heart eyes")
    keywords = ("face", "grin", "heart", "love", "eyes", "emoji")

    def build(self) -> None:
        self.add_arc("face-1", (4, 24), (44, 24), radius_x=20)
        self.add_arc("face-2", (44, 24), (4, 24), radius_x=20)
        self.add_contour("face", "face-1", "face-2", closed=True)
        for index, cx in enumerate((17, 31), 1):
            self.heart(f"eye-{index}", cx)
        self.add_line("grin-1", (16, 32), (32, 32))
        self.add_bezier("grin-2", (32, 32), ((30, 37), (18, 37), (16, 32)))
        self.add_contour("grin", "grin-1", "grin-2", closed=True)

    def heart(self, name, cx, dip=19, bottom=24, half=3):
        self.add_bezier(name, (cx, bottom),
                        ((cx + 1.5, bottom - 1.5), (cx + half, bottom - 3), (cx + half, dip)),
                        ((cx + half, dip - 2.5), (cx, dip - 2.5), (cx, dip)),
                        ((cx, dip - 2.5), (cx - half, dip - 2.5), (cx - half, dip)),
                        ((cx - half, bottom - 3), (cx - 1.5, bottom - 1.5), (cx, bottom)))
        self.add_contour(f"{name}-outline", name, closed=True)
