from icon_set.model.keyshapes import Keyshape
from icon_set.model.icons.solo._base import Solo48
SOURCE_ICON_ID='236451ca-3512-494c-8c17-87481ad251f8'

class Drawing(Solo48):
    icon_id='face-awesome'
    keyshape=Keyshape.HRECT_L
    semantic_role="MAIN"
    semantic_kind="noun"
    category="objects/general"
    aliases=()
    keywords=('face', 'awesome')
    def build(self):
        self.path('head',(4,24),[('A',(44,24),20,16,True),('A',(4,24),20,16,True)],True)
        for i,x in enumerate((16,32)):
            self.add_polyline(f'star-{i}',(x,16),(x+2,20),(x+6,20),(x+3,24),(x+4,28),(x,25),(x-4,28),(x-3,24),(x-6,20),(x-2,20),closed=True)

    def circle(self,n,x,y,r):
        self.add_arc(n+'-a',(x-r,y),(x+r,y),radius_x=r)
        self.add_arc(n+'-b',(x+r,y),(x-r,y),radius_x=r)
        self.add_contour(n,n+'-a',n+'-b',closed=True)
    def path(self,n,start,ops,closed=False):
        at=start; members=[]
        for i,op in enumerate(ops):
            eid=f'{n}-{i}';kind,end,*args=op
            if end==at:continue
            if kind=='L':self.add_line(eid,at,end)
            elif kind=='A':self.add_arc(eid,at,end,radius_x=args[0],radius_y=args[1],sweep=args[2])
            at=end;members.append(eid)
        self.add_contour(n,*members,closed=closed)
