class Drawing(Solo48):
    icon_id = 'air-purifier-with-two-airflow-curves'
    keyshape = Keyshape.VRECT_L
    semantic_role = 'MAIN'
    semantic_kind = 'noun'
    category = 'objects'
    aliases = ()
    keywords = ('air', 'purifier')

    def circle(self, n, x, y, r):
        self.add_arc(n+'-top', (x-r,y), (x+r,y), radius_x=r)
        self.add_arc(n+'-bottom', (x+r,y), (x-r,y), radius_x=r)
        self.add_contour(n,n+'-top',n+'-bottom',closed=True)

    def box(self,n,l,t,r,b,k=4,top=(),bottom=()):
        p=[(l+k,t),*[(x,t) for x in sorted(top)],(r-k,t),(r,t+k),(r,b-k),(r-k,b),*[(x,b) for x in sorted(bottom,reverse=True)],(l+k,b),(l,b-k),(l,t+k)]
        ids=[]
        for i,(a,z) in enumerate(zip(p,p[1:]+p[:1])):
            if a==z:continue
            name=f'{n}-{i}';ids.append(name)
            if a[0]!=z[0] and a[1]!=z[1]:self.add_arc(name,a,z,radius_x=k)
            else:self.add_line(name,a,z)
        self.add_contour(n,*ids,closed=True)

    def build(self):
        # Matched S-shaped air streams, not small isolated arc dots.
        self.box('body',8,22,40,40,4,bottom=(16,32))
        self.add_line('indicator',(20,31),(28,31))
        for i,x in enumerate((16,32)):
            self.add_bezier(f'air-{i}',(x,4),((x-4,4),(x-4,8),(x,8)),((x+4,8),(x+4,12),(x,12)))
            self.add_line(f'foot-{i}',(x,40),(x,44))
            self.relate('connect',f'foot-{i}','body')
